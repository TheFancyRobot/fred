    Create a shared helper at `tests/unit/helpers/mock-tool-registry.ts`:

    ```typescript
    import type { ToolRegistryLike } from '../../../packages/core/src/agent/factory';
    import type { Tool } from '../../../packages/core/src/tool/tool';

    /**
     * Creates an in-memory mock satisfying ToolRegistryLike.
     * Supports registerTool, getTools, hasTool, and getMissingToolIds.
     */
    export function createMockToolRegistry(): ToolRegistryLike {
      const tools = new Map<string, Tool>();

      return {
        registerTool(tool: Tool): void {
          tools.set(tool.id, tool);
        },
        getTools(ids: string[]): Tool[] {
          return ids.map(id => tools.get(id)).filter((t): t is Tool => t !== undefined);
        },
        hasTool(id: string): boolean {
          return tools.has(id);
        },
        getMissingToolIds(ids: string[]): string[] {
          return ids.filter(id => !tools.has(id));
        },
      };
    }
    ```
