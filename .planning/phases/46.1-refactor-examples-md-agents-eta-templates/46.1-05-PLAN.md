```typescript
// Adds a custom namespace to the template context. The resolver is called
// each time the global variables resolver is updated, producing a snapshot
// that is spread into the ETA body context as a top-level namespace.
addTemplateContext(namespace: string, resolver: () => unknown): void;

// Process a message through the agent pipeline. The system message template
// is resolved per-message, picking up the latest addTemplateContext snapshots.
async processMessage(message: string, options?: ProcessMessageOptions): Promise<AgentResponse | null>;
```

From packages/core/src/template/context.ts (BodyContext):
```typescript
// Custom namespaces from addTemplateContext are spread into BodyContext:
// buildBodyContext(..., customNamespaces) => { vars, env, config, agent, ...customNamespaces }
// So addTemplateContext('session', () => ({ userId: '42' })) makes session.userId
// available in ETA templates as <%= session.userId %>
export type BodyContext = FrontmatterContext & {
  agent: AgentTemplateContext;
} & Record<string, unknown>;
```
